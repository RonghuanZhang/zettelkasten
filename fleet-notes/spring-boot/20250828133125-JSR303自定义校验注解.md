package ai.saharalabs.hive.validator;

import jakarta.validation.Constraint;
import jakarta.validation.Payload;

import java.lang.annotation.*;

@Documented
@Constraint(validatedBy = FieldCompareValidator.class)
@Target({ ElementType.TYPE })
@Retention(RetentionPolicy.RUNTIME)
@Repeatable(FieldCompares.class)
public @interface FieldCompare {
    String message() default "{field.compare.invalid}";
    Class<?>[] groups() default {};
    Class<? extends Payload>[] payload() default {};

    String first();

    String second();

    /** ">", ">=", "<", "<=", "==" */
    String operator() default ">";
}

package ai.saharalabs.hive.validator;

import jakarta.validation.ConstraintValidator;
import jakarta.validation.ConstraintValidatorContext;

import java.lang.reflect.Field;

public class FieldCompareValidator implements ConstraintValidator<FieldCompare, Object> {
    private String firstFieldName;
    private String secondFieldName;
    private String operator;

    @Override
    public void initialize(FieldCompare constraintAnnotation) {
        this.firstFieldName = constraintAnnotation.first();
        this.secondFieldName = constraintAnnotation.second();
        this.operator = constraintAnnotation.operator();
    }

    @Override
    public boolean isValid(Object value, ConstraintValidatorContext constraintValidatorContext) {
        try {
            if (value == null) return true;

            Field firstField = value.getClass().getDeclaredField(firstFieldName);
            Field secondField = value.getClass().getDeclaredField(secondFieldName);
            firstField.setAccessible(true);
            secondField.setAccessible(true);

            Object firstValue = firstField.get(value);
            Object secondValue = secondField.get(value);

            if (firstValue == null || secondValue == null) {
                return true;
            }

            if (!(firstValue instanceof Comparable) || !(secondValue instanceof Comparable)) {
                return false;
            }

            @SuppressWarnings("unchecked")
            Comparable<Object> firstComparable = (Comparable<Object>) firstValue;
            int cmp = firstComparable.compareTo(secondValue);

            return switch (operator) {
                case ">"  -> cmp > 0;
                case ">=" -> cmp >= 0;
                case "<"  -> cmp < 0;
                case "<=" -> cmp <= 0;
                case "==" -> cmp == 0;
                default   -> throw new IllegalArgumentException("Unsupported operator: " + operator);
            };

        } catch (Exception e) {
            return false;
        }
    }
}
