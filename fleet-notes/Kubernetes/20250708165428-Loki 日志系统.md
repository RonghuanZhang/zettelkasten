## Loki API


```shell

# 查看所有的标签
curl -s "http://127.0.0.1:3100/loki/api/v1/labels" \
     -H 'X-Scope-OrgID: ronghzhang11' \
  | jq .

# 查看标签的值
curl -s "http://127.0.0.1:3100/loki/api/v1/label/namespace/values" \
     -H 'X-Scope-OrgID: ronghzhang11' \
  | jq .
  
# 查看日志内容
curl "http://127.0.0.1:3100/loki/api/v1/query_range" --data-urlencode 'query={namespace="tenant_id"}' -H 'X-Scope-OrgId: ronghzhang11'

# 查看Knative Service 日志
curl "http://34.30.64.191/loki/api/v1/query_range" --data-urlencode 'query={namespace="ronghzhang11", knative_configuration="rhz-hive-envs-sample-01",detected_level="warn"} ' --data-urlencode 'since=36h' -H 'X-Scope-OrgId: ronghzhang11'

curl "http://34.30.64.191/loki/api/v1/query_range" --data-urlencode 'query={namespace="ronghzhang11", knative_configuration="rhz-hive-envs-sample-01",env="staging"} |= "WARNING" |= "Creating an Engine" ' --data-urlencode 'since=12h' -H 'X-Scope-OrgId: ronghzhang11'

|= 简单子串匹配
|~ "timeout|timed out" 正则表达式匹配，Go RE2 正则表达式语法


# 查看pattern
curl "http://34.55.125.35/loki/api/v1/patterns" --data-urlencode 'query={namespace="ronghzhang11",knative_configuration="rhz-hive-envs-sample-01"}' -H 'X-Scope-OrgId: ronghzhang11'

# tail loki
websocat "ws://127.0.0.1:3100/loki/api/v1/tail?query={namespace=\"ronghzhang11\",knative_configuration=\"rhz-hive-envs-sample-01\"}&limit=100" \
  -H 'X-Scope-OrgId: ronghzhang11'

# 查看 Ring 结果
curl "http://34.30.64.191/ring" -H 'X-Scope-OrgId: ronghzhang11' > ring.html
```
